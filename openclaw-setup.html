Fern 계정에서 LaunchAgent를 등록하여 OpenClaw Gateway/Browser를 자동 생성, 재시작, 상태 검증합니다.

별도 계정·운영 방식이므로 아래 섹션에서 run-openclaw.sh를 LaunchAgent로 관리하세요.

#!/bin/bash
# ==============================================================================
# OpenClaw Agent Launcher (run-openclaw.sh)
# 용도: 에이전트의 LaunchAgent 기반 구동 및 생존성 관리
# 위치: /Users/fern/run-openclaw.sh
# 실행: 에이전트 계정(fern)에서 실행
# ==============================================================================

set -euo pipefail

CMD="${1:-status}"

SANDBOX_DIR="/Users/fern/sandbox"
LOG_DIR="${SANDBOX_DIR}/logs"
TMPDIR="${LOG_DIR}/tmp"
GATEWAY_PORT="18789"
OPENCLAW_BIN="/Users/fern/.nvm/versions/node/v24.13.1/bin/openclaw"
PLIST_PATH="/Users/fern/Library/LaunchAgents/com.fern.openclaw.plist"

verify_mount() {
  MOUNT_POINT="/Users/fern/sandbox"
  EXPECTED_VOLUME="FernSandbox"
  DISKUTIL="/usr/sbin/diskutil"
  for i in $(seq 1 15); do
    if INFO="$($DISKUTIL info "$MOUNT_POINT" 2>/dev/null)"; then
      VOLUME_NAME="$(echo "$INFO" | awk -F': *' '/Volume Name/ {print $2}')"
      MOUNT_PATH="$(echo "$INFO" | awk -F': *' '/Mount Point/ {print $2}')"
      if [ "$VOLUME_NAME" = "$EXPECTED_VOLUME" ] && [ "$MOUNT_PATH" = "$MOUNT_POINT" ]; then
        return 0
      fi
    fi
    sleep 2
  done
  echo "[ERROR] Sandbox mount not ready."
  exit 78
}

cleanup() {
  pkill -u fern -f "Google Chrome" 2>/dev/null || true
  pkill -u fern -f "Chromium" 2>/dev/null || true
  pkill -u fern -f "openclaw gateway" 2>/dev/null || true
  pkill -u fern -f "openclaw dashboard" 2>/dev/null || true
  pkill -u fern -f "openclaw relay" 2>/dev/null || true
  PIDS=$(/usr/sbin/lsof -nP -iTCP:${GATEWAY_PORT} -sTCP:LISTEN -u fern -t 2>/dev/null || true)
  for pid in $PIDS; do
    if [ "$pid" != "$$" ]; then kill -9 "$pid" 2>/dev/null || true; fi
  done
}

daemon() {
  verify_mount
  mkdir -p "${LOG_DIR}" "${TMPDIR}"
  exec >> "${LOG_DIR}/openclaw-launchd.out" 2>&1
  echo "=================================================="
  echo "🚀 OpenClaw DAEMON START: $(date)"
  echo "=================================================="
  export NVM_DIR="/Users/fern/.nvm"
  if [ -s "$NVM_DIR/nvm.sh" ]; then
    set +u
    . "$NVM_DIR/nvm.sh"
    set -u
    nvm use 24 >/dev/null 2>&1 || true
  fi
  cd "${SANDBOX_DIR}"
  cleanup
  sleep 1
  if [ ! -x "$OPENCLAW_BIN" ]; then
    echo "[ERROR] openclaw binary not found."
    sleep 10
    exit 127
  fi
  (sleep 3 && "$OPENCLAW_BIN" dashboard) &
  exec "$OPENCLAW_BIN" gateway --bind loopback --port "${GATEWAY_PORT}"
}

install_plist() {
  mkdir -p /Users/fern/Library/LaunchAgents
  cat > "$PLIST_PATH" << PLIST_EOF




    Labelcom.fern.openclaw
    ProgramArguments/Users/fern/run-openclaw.shdaemon
    RunAtLoadKeepAliveThrottleInterval10
    LimitLoadToSessionTypeAqua
    EnvironmentVariables
    
        PATH/Users/fern/.nvm/versions/node/v24.13.1/bin:/usr/bin:/bin:/usr/sbin:/sbin
    


PLIST_EOF
}

start() {
  UID_FERN=$(id -u)
  install_plist
  launchctl bootout "gui/${UID_FERN}" "$PLIST_PATH" 2>/dev/null || true
  sleep 1
  launchctl bootstrap "gui/${UID_FERN}" "$PLIST_PATH"
  echo "✅ LaunchAgent 시작 완료"
}

stop() {
  UID_FERN=$(id -u)
  launchctl bootout "gui/${UID_FERN}" "$PLIST_PATH" 2>/dev/null || true
  sleep 1
  cleanup
  echo "✅ 완전 종료 완료"
}

status() {
  UID_FERN=$(id -u)
  echo "=== OpenClaw 상태 ==="
  if launchctl print "gui/${UID_FERN}/com.fern.openclaw" >/dev/null 2>&1; then echo "🟢 LaunchAgent 활성"; else echo "🔴 LaunchAgent 중지"; fi
  if lsof -nP -iTCP:${GATEWAY_PORT} -sTCP:LISTEN >/dev/null 2>&1; then echo "🟢 Gateway LISTEN (${GATEWAY_PORT})"; else echo "🔴 Gateway 닫힘"; fi
}

restart() { stop; sleep 2; start; }

case "$CMD" in
  start) start ;; stop) stop ;; restart) restart ;; status) status ;; daemon) daemon ;;
  *) echo "Usage: $0 {start|stop|restart|status}"; exit 1 ;;
esac
RUNNER_EOF

chmod 700 /Users/fern/run-openclaw.sh
echo "✅ run-openclaw.sh 생성 및 권한 설정 완료!"

OpenClaw 전체 스택 초기화 가이드

아래 절차는 되돌릴 수 없는 삭제를 포함합니다. 단계를 순서대로 실행하세요. 부분 제거가 필요하면 해당 섹션만 선택 실행하세요.

Step 1 — 프로세스 중지 & 서비스 해제

실행 중인 AI 프로세스를 모두 종료하고 Ollama 및 자동 업데이트 서비스를 언로드합니다.

# Ollama 및 자동 업데이트 데몬 언로드
sudo launchctl bootout system /Library/LaunchDaemons/com.alvin.ollama.plist 2>/dev/null || true
launchctl bootout gui/$(id -u fern) /Users/fern/Library/LaunchAgents/com.fern.npmupdate.plist 2>/dev/null || true

# Fern 프로세스 및 잔류 노드 종료
sudo pkill -9 -u fern -f openclaw
sudo pkill -9 -u fern node
pkill -f 'ollama serve' 2>/dev/null || true

# 포트 점검
sudo lsof -nP -iTCP:11434,18789,18792 -sTCP:LISTEN

Step 2 — OpenClaw & 플러그인 제거 필수

Fern 계정에서 openclaw, 플러그인, 설정 파일을 제거합니다.

# Fern 계정에서 실행
su - fern

# OpenClaw 본체 제거
npm uninstall -g openclaw

# OpenClaw 설정 파일 삭제 (영구 삭제)
rm -rf ~/.openclaw

# Playwright 브라우저 바이너리 삭제
npx playwright uninstall --all 2>/dev/null || true
npm uninstall -g @playwright/test

# Alvin 계정으로 복귀
exit

Step 3 — Fern 계정 & 홈 디렉터리 삭제

계정 삭제 전 FernSandbox 볼륨을 먼저 언마운트합니다.

# 1. 샌드박스 볼륨 언마운트
sudo diskutil unmount /Users/fern/sandbox 2>/dev/null || true

# 2. fern 계정 및 홈 디렉터리 삭제
sudo sysadminctl -deleteUser fern
sudo rm -rf /Users/fern

# 3. 삭제 확인
dscl . list /Users | grep fern   # 아무것도 출력되지 않으면 정상

Step 4 — FernSandbox APFS 볼륨 & fstab 제거 

# 1. 볼륨이 남아 있다면 강제 언마운트 후 삭제
diskutil list | grep FernSandbox   # 디스크 식별자 확인 (예: disk3s5)
sudo diskutil apfs deleteVolume FernSandbox

# 2. fstab에서 FernSandbox UUID 항목 제거
# 아래 명령으로 해당 줄만 삭제 (UUID는 실제 값으로 치환)
VOL_UUID=$(diskutil info FernSandbox 2>/dev/null | awk '/Volume UUID:/ {print $3}')
[[ -n "$VOL_UUID" ]] && sudo sed -i '' "/UUID=$VOL_UUID/d" /etc/fstab

# 3. fstab 결과 확인
cat /etc/fstab

Step 5 — PF 방화벽 & Anchor 제거 

# 1. pf.conf에서 fern anchor 라인 제거
sudo sed -i '' '/anchor.*fern/d' /etc/pf.conf
sudo sed -i '' '/load anchor.*fern/d' /etc/pf.conf

# 2. Anchor 파일 삭제
sudo rm -f /etc/pf.anchors/fern

# 3. PF 규칙 리로드 (기본 상태로)
sudo pfctl -f /etc/pf.conf

# 4. 확인 (fern 관련 룰이 없으면 정상)
sudo pfctl -sr | grep fern   # 아무것도 출력되지 않으면 정상

Step 6 — SSH 정책 & 홈 디렉터리 권한 복원 선택 

# sshd_config의 Fern 차단 규칙 제거 (필요 시)
sudo sed -i '' '/^DenyUsers fern$/d' /etc/ssh/sshd_config

# Alvin 홈 디렉터리 권한 원복 (필요 시)
# sudo chmod 755 /Users/alvin

# /Users/Shared ACL 초기화 (fern deny 항목 제거)
sudo chmod -a "user:fern deny list,add_file,search,delete,add_subdirectory,writeattr,writeextattr,chown,file_inherit,directory_inherit" /Users/Shared 2>/dev/null || true

# 모델 캐시 삭제 (~/Library/Application Support/ollama)
rm -rf ~/Library/Application\ Support/ollama

SEC 2 에이전트 런너 (`run-openclaw.sh`) & LaunchAgent 설정

에이전트 계정(fern)의 로그인 시 자동으로 구동되는 런너 스크립트와 macOS LaunchAgent 서비스를 구축합니다.
실행: 인프라 구축 완료 후 su - fern 계정으로 전환하여 실행하세요. 

#!/bin/bash
# ==============================================================================
# OpenClaw Agent Runner & LaunchAgent Installer
# ==============================================================================
USER_HOME="/Users/fern"
SANDBOX_DIR="${USER_HOME}/sandbox"
LOG_DIR="${SANDBOX_DIR}/logs"
TMP_DIR="${LOG_DIR}/tmp"
RUNNER_SCRIPT="${USER_HOME}/run-openclaw.sh"
PLIST_NAME="com.fern.openclaw"
PLIST_PATH="${USER_HOME}/Library/LaunchAgents/${PLIST_NAME}.plist"

mkdir -p "${LOG_DIR}" "${TMP_DIR}" "${USER_HOME}/Library/LaunchAgents"

# 1. run-openclaw.sh 생성
cat > "${RUNNER_SCRIPT}" << 'RUNNER_EOF'
#!/bin/bash
set -euo pipefail
if ! mount | grep -q "on /Users/fern/sandbox "; then exit 78; fi
export SANDBOX_DIR="/Users/fern/sandbox"
export LOG_DIR="${SANDBOX_DIR}/logs"
export TMPDIR="${LOG_DIR}/tmp"
export OPENCLAW_GATEWAY_BIND="127.0.0.1"
export GATEWAY_PORT="18789"
export NVM_DIR="/Users/fern/.nvm"
source "${NVM_DIR}/nvm.sh"
pkill -u fern -f "Google Chrome" 2>/dev/null || true
(sleep 3 && openclaw dashboard) &
exec openclaw gateway --bind "${OPENCLAW_GATEWAY_BIND}" --port "${GATEWAY_PORT}"
RUNNER_EOF
chmod 700 "${RUNNER_SCRIPT}"

# 2. LaunchAgent plist 생성
cat > "${PLIST_PATH}" << PLIST_EOF




    Label${PLIST_NAME}
    ProgramArguments${RUNNER_SCRIPT}
    RunAtLoad
    KeepAlive
    LimitLoadToSessionTypeAqua
    StandardOutPath${LOG_DIR}/openclaw-launchd.out
    StandardErrorPath${LOG_DIR}/openclaw-launchd.err


PLIST_EOF

# 3. start-agent.sh / stop-agent.sh (Control Scripts)
cat > "${USER_HOME}/start-agent.sh" << 'START_EOF'
#!/bin/bash
PLIST_PATH="/Users/fern/Library/LaunchAgents/com.fern.openclaw.plist"
launchctl unload -w "${PLIST_PATH}" 2>/dev/null || true
launchctl load -w "${PLIST_PATH}"
echo "✅ LaunchAgent started. Log: tail -f /Users/fern/sandbox/logs/openclaw-launchd.out"
START_EOF
chmod 700 "${USER_HOME}/start-agent.sh"

cat > "${USER_HOME}/stop-agent.sh" << 'STOP_EOF'
#!/bin/bash
PLIST_PATH="/Users/fern/Library/LaunchAgents/com.fern.openclaw.plist"
launchctl unload -w "${PLIST_PATH}" 2>/dev/null || true
pkill -u fern -f "Google Chrome" 2>/dev/null || true
echo "✅ LaunchAgent stopped."
STOP_EOF
chmod 700 "${USER_HOME}/stop-agent.sh"

echo "🎉 Agent Runner and LaunchAgent setup complete."

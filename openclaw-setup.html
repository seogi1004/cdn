🛡️ 핵심 4: PF 네트워크 격리 (Anchor 상세 v3.4) 필수

💡 v3.4 Anchor 요약: VPN + RFC1918 전면 차단 필수 

# 1. 전용 Anchor 파일 생성 (/etc/pf.anchors/fern)
sudo tee /etc/pf.anchors/fern << 'EOF'
# 1) lo0 예외 (11434: Ollama / 18792: API Gateway, no state)
pass out quick on lo0 proto tcp from any to 127.0.0.1 port 11434 user fern no state
pass out quick on lo0 proto tcp from 127.0.0.1 port 11434 to any user fern no state

pass out quick on lo0 proto tcp from any to 127.0.0.1 port 18792 user fern no state
pass out quick on lo0 proto tcp from 127.0.0.1 port 18792 to any user fern no state

# 2) 나머지 로컬호스트 차단
block drop out quick proto { tcp, udp } from any to 127.0.0.1 user fern

# 3) VPN 및 사설 IP 차단
block drop out quick on { utun0, utun1, utun2, utun3, utun4, ipsec0 } proto { tcp, udp } user fern
block drop out quick proto { tcp, udp } from any to 10.0.0.0/8 user fern
block drop out quick proto { tcp, udp } from any to 172.16.0.0/12 user fern
block drop out quick proto { tcp, udp } from any to 192.168.0.0/16 user fern

# 4) mDNS 및 IPv6 내부망 차단
block drop out quick proto udp from any to any port 5353 user fern
block drop out quick inet6 proto { tcp, udp } from any to fe80::/10 user fern
block drop out quick inet6 proto { tcp, udp } from any to fc00::/7 user fern
EOF

# 문법 검증 및 리로드
if sudo pfctl -n -f /etc/pf.conf >/dev/null 2>&1; then
  sudo pfctl -f /etc/pf.conf && sudo pfctl -e
  echo '✅ PF 방화벽 적용 완료 (활성화됨)'
else
  echo '🚨 문법 에러가 있습니다. 로그를 확인하세요.'
fi

🚀

보안 마스터 스크립트 (Verified V3.4)

안정성

VPN/사설망 차단 강화 + fstab 영구 마운트 + ulimit 강제 종속 통합 버전

#!/bin/bash
set -euo pipefail

# OpenClaw M4 Max Security Master Script (Verified & Refined V3.4)
ALVIN_HOME="/Users/alvin"
FERN_HOME="/Users/fern"
SANDBOX_MP="/Users/fern/sandbox"
SANDBOX_VOL="FernSandbox"
SANDBOX_QUOTA="100g"

echo ">>> [1/9] 홈 디렉토리 격리..."
sudo chmod 700 "$ALVIN_HOME" "$FERN_HOME"

echo ">>> [2/9] ACL 차단 적용 (성능/안정성 최적화)..."
sudo chmod +a "user:fern deny list,add_file,search,delete,add_subdirectory,writeattr,writeextattr,chown,file_inherit,directory_inherit" /Users/Shared || true
sudo chmod +a "user:fern deny list,search,add_file,add_subdirectory,file_inherit,directory_inherit" /Volumes/* >/dev/null 2>&1 || true

echo ">>> [3/9] APFS 100GB 샌드박스 볼륨 생성 및 fstab 영구 마운트..."
CID="$(diskutil info / | awk -F': *' '/APFS Container:/ {print $2; exit}')"
if [[ -z "${CID}" ]]; then echo "ERROR: APFS Container 감지 실패"; exit 1; fi

sudo mkdir -p "$SANDBOX_MP"
if ! diskutil list | grep -q "$SANDBOX_VOL"; then
  sudo diskutil apfs addVolume "$CID" APFS "$SANDBOX_VOL" -quota "$SANDBOX_QUOTA"
fi

VOL_UUID=$(diskutil info "$SANDBOX_VOL" | awk '/Volume UUID:/ {print $3}')
if [[ -n "$VOL_UUID" ]] && ! grep -q "$VOL_UUID" /etc/fstab 2>/dev/null; then
  echo "UUID=$VOL_UUID $SANDBOX_MP apfs rw,nodev,nosuid" | sudo tee -a /etc/fstab >/dev/null
fi

sudo diskutil unmount "$SANDBOX_VOL" >/dev/null 2>&1 || true
if ! sudo diskutil mount -mountPoint "$SANDBOX_MP" "$SANDBOX_VOL"; then
  echo "ERROR: FernSandbox 마운트 실패! 볼륨 상태를 확인하세요." && exit 1
fi
sudo chown fern:staff "$SANDBOX_MP" && sudo chmod 700 "$SANDBOX_MP"

echo ">>> [4/9] SSH 원격 접속 원천 봉쇄..."
sudo dseditgroup -o create com.apple.access_ssh >/dev/null 2>&1 || true
sudo dseditgroup -o edit -a alvin -t user com.apple.access_ssh
sudo dseditgroup -o edit -d fern -t user com.apple.access_ssh >/dev/null 2>&1 || true

echo ">>> [5/9] PF 방화벽 정책 반영..."
sudo mkdir -p /etc/pf.anchors
sudo tee /etc/pf.anchors/fern >/dev/null <<'EOF'
# 1) lo0 예외 (11434: Ollama / 18792: API Gateway, no state)
pass out quick on lo0 proto tcp from any to 127.0.0.1 port 11434 user fern no state
pass out quick on lo0 proto tcp from 127.0.0.1 port 11434 to any user fern no state

pass out quick on lo0 proto tcp from any to 127.0.0.1 port 18792 user fern no state
pass out quick on lo0 proto tcp from 127.0.0.1 port 18792 to any user fern no state

# 2) 나머지 로컬호스트 차단
block drop out quick proto { tcp, udp } from any to 127.0.0.1 user fern

# 3) VPN 및 사설 IP 차단
block drop out quick on { utun0, utun1, utun2, utun3, utun4, ipsec0 } proto { tcp, udp } user fern
block drop out quick proto { tcp, udp } from any to 10.0.0.0/8 user fern
block drop out quick proto { tcp, udp } from any to 172.16.0.0/12 user fern
block drop out quick proto { tcp, udp } from any to 192.168.0.0/16 user fern

# 4) mDNS 및 IPv6 내부망 차단
block drop out quick proto udp from any to any port 5353 user fern
block drop out quick inet6 proto { tcp, udp } from any to fe80::/10 user fern
block drop out quick inet6 proto { tcp, udp } from any to fc00::/7 user fern
EOF

if ! grep -q 'anchor "fern"' /etc/pf.conf; then
  echo 'anchor "fern"' | sudo tee -a /etc/pf.conf >/dev/null
  echo 'load anchor "fern" from "/etc/pf.anchors/fern"' | sudo tee -a /etc/pf.conf >/dev/null
fi

if sudo pfctl -n -f /etc/pf.conf >/dev/null 2>&1; then
  sudo pfctl -f /etc/pf.conf && sudo pfctl -e
  echo '✅ PF 방화벽 적용 완료 (활성화됨)'
else
  echo '🚨 문법 에러가 있습니다. 로그를 확인하세요.'
fi

echo ">>> [6/9] Ollama 가동 (PM2 - Admin)..."
pm2 delete ollama >/dev/null 2>&1 || true
pm2 start --name ollama --interpreter bash -- bash -lc 'export OLLAMA_HOST=127.0.0.1:11434; ollama serve'
pm2 save

echo ">>> [7/9] FernBot 가동 및 ulimit 강제 종속..."
sudo -H -u fern env -C /Users/fern zsh -lc 'pm2 delete fern-bot >/dev/null 2>&1 || true'
sudo -H -u fern env -C /Users/fern zsh -lc 'pm2 start zsh --name fern-bot -- -lc "ulimit -u 512 && cd /Users/fern/sandbox && openclaw gateway"'
sudo -H -u fern env -C /Users/fern zsh -lc 'pm2 save'

echo ">>> [8/9] ulimit 제한 설정..."
sudo -H -u fern env -C /Users/fern zsh -lc "grep -q '^ulimit -u 512' '${FERN_HOME}/.zshrc' 2>/dev/null || echo 'ulimit -u 512' >> '${FERN_HOME}/.zshrc'"

echo ">>> [9/9] 보안 검증 결과 출력..."
echo ">>> [DONE] V3.4 보안 마스터 배포 완료"

Step 3 — Fern 계정 & 홈 디렉터리 삭제

Step 4 — FernSandbox APFS 볼륨 & fstab 제거 

# 1. 볼륨이 남아 있다면 강제 언마운트 후 삭제
diskutil list | grep FernSandbox   # 디스크 식별자 확인 (예: disk3s5)
sudo diskutil apfs deleteVolume FernSandbox

# 2. fstab에서 FernSandbox UUID 항목 제거
# 아래 명령으로 해당 줄만 삭제 (UUID는 실제 값으로 치환)
VOL_UUID=$(diskutil info FernSandbox 2>/dev/null | awk '/Volume UUID:/ {print $3}')
[[ -n "$VOL_UUID" ]] && sudo sed -i '' "/UUID=$VOL_UUID/d" /etc/fstab

# 3. fstab 결과 확인
cat /etc/fstab

Step 5 — PF 방화벽 & Anchor 제거

sudo sed -i '' '/load anchor.*fern/d' /etc/pf.conf

# 2. Anchor 파일 삭제
sudo rm -f /etc/pf.anchors/fern

# 3. PF 규칙 리로드 (기본 상태로)

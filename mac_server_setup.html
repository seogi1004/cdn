1. 계정 및 시스템 인프라 (Alvin)

Alvin 계정에서 전체 시스템을 관리하고 봇 계정을 생성합니다. 권한 격리를 통해 봇이 시스템을 망가뜨리는 것을 방지합니다.

Terminal (Alvin) 
# 1. 봇 계정 생성 (Password: fern1234, jen1234)
sudo sysadminctl -addUser fern -fullName "Fern Personal" -password "fern1234" -home /Users/fern
sudo sysadminctl -addUser jennifer -fullName "Jennifer Team" -password "jen1234" -home /Users/jennifer

# 2. 홈 디렉토리 권한 초기화 (필수)
# 계정 생성 직후 권한 꼬임을 방지합니다.
sudo mkdir -p /Users/fern/.ssh && sudo chown fern:staff /Users/fern/.ssh && sudo chmod 700 /Users/fern/.ssh
sudo mkdir -p /Users/jennifer/.ssh && sudo chown jennifer:staff /Users/jennifer/.ssh && sudo chmod 700 /Users/jennifer/.ssh

# 3. 권한 격리 (각자의 방은 각자만 출입)
sudo chmod 700 /Users/fern
sudo chmod 750 /Users/jennifer

# 4. 절전 방지 (서버 모드 활성화)
sudo pmset -a sleep 0 && sudo pmset -a displaysleep 15 && sudo pmset -a wmp 1

3. AI 인프라 구축 (Alvin)

Alvin 계정에서 Ollama를 시스템 데몬으로 구동하여, 모든 봇이 로컬 AI 자원을 공유하도록 설정합니다.

Terminal (Alvin) 
# 1. 필수 도구 설치 (Homebrew, Ollama, PM2)
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
brew install ollama zellij eza bat fzf zoxide btop
npm install -g pm2

# 2. Ollama 실행 (System-wide Daemon)
# - Parallel 4: Fern이 여러 작업을 동시에 요청해도 처리 가능
# - Max Loaded 2: 텍스트(DeepSeek)와 비전(Llava) 동시 메모리 로드
pm2 start "ollama serve" --name ollama-core \
  --env OLLAMA_HOST="0.0.0.0:11434" \
  --env OLLAMA_ORIGINS="*" \
  --env OLLAMA_NUM_PARALLEL="4" \
  --env OLLAMA_MAX_LOADED_MODELS="2" \
  --env OLLAMA_KEEP_ALIVE="24h"

# 3. 모델 다운로드 (Alvin이 받으면 Fern도 사용 가능)
ollama pull deepseek-r1:32b
ollama pull llava:34b

pm2 save && pm2 startup

6. 봇 코드 생성 (The Brains)

Vital Step

CLI 설치가 완료되면, 아래 코드를 복사하여 각 계정의 ~/openclaw/agent.js (또는 fern.js/jennifer.js)로 저장하세요.

Fern.js

~/openclaw/fern.js 
const { chromium } = require('playwright');
const axios = require('axios');
require('dotenv').config();

async function runFern() {
  // [Dynamic Port] 환경변수 또는 기본값
  const port = process.env.TARGET_PORT || '3000';
  const targetUrl = `http://localhost:${port}`;
  console.log(`🌿 Visiting: ${targetUrl}`);

  // [Persistent Profile] 로그인 유지
  const context = await chromium.launchPersistentContext(
    process.env.FERN_PROFILE, 
    { headless: false }
  );
  const page = await context.newPage();

  try {
      await page.goto(targetUrl);
      console.log("✅ Connected to Localhost!");
      
      // DOM 추출 -> Local DeepSeek 질의 로직
      // ...
  } catch (e) {
      console.error(`❌ Connection failed: ${targetUrl}`);
  }
}
runFern();

Jennifer.js

~/openclaw/jennifer.js 
const { chromium } = require('playwright');
const Anthropic = require('@anthropic-ai/sdk');
require('dotenv').config();

const anthropic = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY 
});

async function runJennifer() {
  // [Dynamic URL]
  const targetUrl = process.env.TARGET_URL || 'https://dev.jennifersoft.com';
  console.log(`⭐️ Scouting: ${targetUrl}`);

  const browser = await chromium.launch();
  const page = await browser.newPage();
  await page.goto(targetUrl);

  // Vision Logic
  const screenshot = await page.screenshot();
  
  // Claude 4.5 Opus (Frontier Model)
  const msg = await anthropic.messages.create({
    model: "claude-opus-4-5-20251101",
    max_tokens: 2048,
    messages: [{
      role: "user",
      content: [
        { type: "image", source: { type: "base64", media_type: "image/jpeg", data: screenshot.toString('base64') }},
        { type: "text", text: "Check UI bugs strictly." }
      ]
    }]
  });
  console.log(msg.content[0].text);
}
runJennifer();

7. Slack 연동 (Socket Mode)

방화벽 내부의 Mac Server가 외부와 소통하는 유일한 창구입니다.

  1. Slack API Console: App 생성 -> Settings -> Socket Mode 활성화 (`xapp-...` 토큰 확보)
  2. OAuth & Permissions: Bot Token Scopes에 app_mentions:read, chat:write 추가.
  3. Install to Workspace: 설치 후 Bot Token (`xoxb-...`) 획득.
